-- Q1
select * from pedidos where status='PAGO';

--Q2	
select id_pedido, data_pedido, status from pedidos
	order by data_pedido DESC limit 3;

--Q3
select p.* from pedidos p
	inner join enderecos e ON p.id_endereco_entrega = e.id_endereco
where e.uf = 'SP' and p.status = 'ENTREGUES';

--Q4
select 
	p.id_pedido
from pedidos p
inner join clientes c on c.id_cliente = p.id_cliente
where c.nome='Ana Souza';

-- Q5
select
	ped.id_pedido
from pedidos ped
inner join
	itens_pedido ip on ip.id_pedido = ped.id_pedido
inner join 
	produtos p on p.id_produto = ip.id_produto
inner join 
	categorias c on c.id_categoria = p.id_categoria
where
	c.nome = 'Livros';
